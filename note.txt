// src/context/TenantContext.tsx
import { createContext, useContext, useEffect, useState } from 'react';
import { getTenantFromUrl } from '@/utils/tenantHelpers';
import { getSchoolBySlug } from '@/services/tenantService';

interface School {
  id: string;
  name: string;
  slug: string;
  logo: string;
  primaryColor?: string;
  secondaryColor?: string;
}

interface TenantContextType {
  school: School | null;
  isLoading: boolean;
  isMainSite: boolean;
}

const TenantContext = createContext<TenantContextType | undefined>(undefined);

export function TenantProvider({ children }: { children: React.ReactNode }) {
  const [school, setSchool] = useState<School | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [isMainSite, setIsMainSite] = useState(false);

  useEffect(() => {
    const loadSchool = async () => {
      const subdomain = getTenantFromUrl();
      
      if (!subdomain) {
        // Main site (tlearn.com)
        setIsMainSite(true);
        setIsLoading(false);
        return;
      }

      try {
        // Fetch school data from backend
        const schoolData = await getSchoolBySlug(subdomain);
        setSchool(schoolData);
        
        // Optionally apply school's brand colors
        if (schoolData.primaryColor) {
          document.documentElement.style.setProperty('--school-primary', schoolData.primaryColor);
        }
      } catch (error) {
        console.error('School not found:', error);
        // Redirect to 404 or main site
      } finally {
        setIsLoading(false);
      }
    };

    loadSchool();
  }, []);

  return (
    <TenantContext.Provider value={{ school, isLoading, isMainSite }}>
      {children}
    </TenantContext.Provider>
  );
}

export const useTenant = () => {
  const context = useContext(TenantContext);
  if (!context) throw new Error('useTenant must be used within TenantProvider');
  return context;
};

// src/App.tsx
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import { TenantProvider, useTenant } from '@/context/TenantContext';

// Main site pages
import LandingPage from '@/pages/public/LandingPage';
import SuperAdminLogin from '@/pages/super-admin/LoginPage';
import SuperAdminDashboard from '@/pages/super-admin/Dashboard';

// School pages
import StudentLogin from '@/pages/school/StudentLogin';
import StudentRegister from '@/pages/school/StudentRegister';
import SchoolAdminLogin from '@/pages/school/AdminLogin';
import StudentDashboard from '@/pages/student/Dashboard';
import SchoolAdminDashboard from '@/pages/admin/Dashboard';

function AppRoutes() {
  const { school, isMainSite, isLoading } = useTenant();

  if (isLoading) {
    return <div>Loading...</div>;
  }

  // ============================================
  // MAIN SITE (tlearn.com)
  // ============================================
  if (isMainSite) {
    return (
      <Routes>
        {/* Public routes */}
        <Route path="/" element={<LandingPage />} />
        <Route path="/about" element={<AboutPage />} />
        <Route path="/pricing" element={<PricingPage />} />
        
        {/* Hidden super admin route */}
        <Route path="/admin-portal" element={<SuperAdminLogin />} />
        
        {/* Super admin dashboard */}
        <Route 
          path="/super-admin/*" 
          element={
            <ProtectedRoute role="SUPER_ADMIN">
              <SuperAdminDashboard />
            </ProtectedRoute>
          } 
        />
      </Routes>
    );
  }

  // ============================================
  // SCHOOL SUBDOMAIN (greenfield.tlearn.com)
  // ============================================
  return (
    <Routes>
      {/* Default: Student login (branded) */}
      <Route path="/" element={<StudentLogin school={school} />} />
      
      {/* Student registration */}
      <Route path="/register" element={<StudentRegister school={school} />} />
      
      {/* Hidden school admin login */}
      <Route path="/admin-login" element={<SchoolAdminLogin school={school} />} />
      
      {/* Student dashboard */}
      <Route 
        path="/student/*" 
        element={
          <ProtectedRoute role="STUDENT">
            <StudentDashboard />
          </ProtectedRoute>
        } 
      />
      
      {/* School admin dashboard */}
      <Route 
        path="/admin/*" 
        element={
          <ProtectedRoute role="ADMIN">
            <SchoolAdminDashboard />
          </ProtectedRoute>
        } 
      />
    </Routes>
  );
}

export default function App() {
  return (
    <BrowserRouter>
      <TenantProvider>
        <AppRoutes />
      </TenantProvider>
    </BrowserRouter>
  );
}

// src/pages/school/StudentLogin.tsx
import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { login } from '@/services/authService';

interface Props {
  school: {
    name: string;
    logo?: string;
    subdomain: string;
  } | null;
}

export default function StudentLogin({ school }: Props) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const navigate = useNavigate();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');
    
    try {
      const data = await login(email, password, school?.subdomain);
      
      if (data.token && data.user.role === 'STUDENT') {
        localStorage.setItem('token', data.token);
        localStorage.setItem('user', JSON.stringify(data.user));
        navigate('/student/dashboard');
      } else {
        setError('Invalid credentials or not a student account');
      }
    } catch (err) {
      setError('Login failed. Please check your credentials.');
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center bg-gray-50">
      <div className="max-w-md w-full bg-white rounded-lg shadow-lg p-8">
        {/* School Logo */}
        {school?.logo && (
          <div className="text-center mb-6">
            <img 
              src={school.logo} 
              alt={school.name} 
              className="h-20 mx-auto"
            />
          </div>
        )}

        {/* School Name */}
        <h1 className="text-2xl font-bold text-center text-[#1F2B7B] mb-2">
          {school?.name || 'School Portal'}
        </h1>
        <p className="text-center text-gray-600 mb-6">
          Student Login
        </p>

        {/* Error Message */}
        {error && (
          <div className="bg-red-100 text-red-700 p-3 rounded mb-4 text-sm">
            {error}
          </div>
        )}

        {/* Login Form */}
        <form onSubmit={handleSubmit}>
          <div className="mb-4">
            <label className="block text-gray-700 mb-2">Email / Student ID</label>
            <input
              type="text"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              className="w-full px-4 py-2 border rounded-lg focus:outline-none focus:border-[#1DA84B]"
              placeholder="Enter your email or student ID"
              required
            />
          </div>

          <div className="mb-6">
            <label className="block text-gray-700 mb-2">Password</label>
            <input
              type="password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              className="w-full px-4 py-2 border rounded-lg focus:outline-none focus:border-[#1DA84B]"
              placeholder="Enter your password"
              required
            />
          </div>

          <button
            type="submit"
            className="w-full bg-[#1DA84B] text-white py-3 rounded-lg font-semibold hover:bg-opacity-90 transition"
          >
            Sign In
          </button>
        </form>

        {/* Register Link */}
        <div className="mt-6 text-center">
          <p className="text-gray-600">
            Don't have an account?{' '}
            <a 
              href="/register" 
              className="text-[#1DA84B] font-semibold hover:underline"
            >
              Register here
            </a>
          </p>
        </div>

        {/* Forgot Password */}

============ COMMON WE LAYOUTS ISSUES & SOLUTIONS ==============
Issue 1: Sidebar Scrolling with Content
Symptom: When you scroll the page, the sidebar moves up/down with the content.
Root Cause: Single scroll container for entire page.
Solution:
tsx// Layout Structure
<div className="flex h-screen overflow-hidden">
  <Sidebar />  {/* Fixed, no scroll */}
  <main className="flex-1 overflow-y-auto">  {/* Scrollable */}
    <Content />
  </main>
</div>
css/* CSS */
html, body {
  height: 100%;
  overflow: hidden;  /* Kill page-level scroll */
}

#root {
  height: 100%;
}
Key Classes:

h-screen = 100vh height
overflow-hidden = No scroll on container
overflow-y-auto = Vertical scroll only on content
flex-1 = Take remaining space


Issue 2: Mobile Zoom on Input Focus
Symptom: On mobile, tapping an input zooms the entire page.
Root Cause: iOS Safari auto-zooms when input font-size < 16px.
Solution:
css/* Force 16px minimum on mobile */
@media screen and (max-width: 768px) {
  input[type="text"],
  input[type="email"],
  input[type="password"],
  input[type="search"],
  textarea,
  select {
    font-size: 16px !important;
  }
}
html<!-- Viewport meta tag -->
<meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=5.0" />
Why 16px? iOS Safari's threshold for auto-zoom. Below 16px = zoom, at/above 16px = no zoom.

Issue 3: Responsive Tables on Mobile
Symptom: Tables too wide, horizontal scroll, bad UX on mobile.
Solution: Show cards on mobile, table on desktop.
tsx{/* Mobile: Cards */}
<div className="grid gap-4 md:hidden">
  {data.map(item => <Card key={item.id}>...</Card>)}
</div>

{/* Desktop: Table */}
<div className="hidden md:block">
  <Table>...</Table>
</div>
Key Classes:

md:hidden = Hide on medium+ screens (show on mobile only)
hidden md:block = Hide on mobile, show on medium+ screens


Quick Debugging Checklist

 Sidebar scrolls? → Check parent has overflow-hidden
 Page zooms on mobile? → Check input font-size >= 16px
 Table too wide? → Use card layout on mobile
 Content not scrolling? → Check overflow-y-auto on content container
 Layout breaking? → Check h-screen and flex setup
